# 田忌赛马问题解题步骤（MD笔记版）

# 核心目标：最大化我方胜利次数

## 一、预处理阶段

1. 输入数据：读取双方元素数量 `n`，以及我方数组 `a`、对方数组 `b`（长度均为 `n`）
2. 排序处理：将我方数组 `a` 和对方数组 `b` 均按 **升序** 排序（统一顺序，为双指针匹配打基础）
3. 双指针初始化：        
   1. 我方弱指针 `i1`：指向我方当前最弱元素，起始值 `0`
   2. 我方强指针 `i2`：指向我方当前最强元素，起始值 `n-1`
   3. 对方弱指针 `j1`：指向对方当前最弱元素，起始值 `0`
   4. 对方强指针 `j2`：指向对方当前最强元素，起始值 `n-1`

## 二、核心贪心匹配（循环核心）

**循环条件**：`i1 <= i2`（确保我方所有元素都完成匹配）

### 情况1：我方最弱 > 对方最弱（`a[i1] > b[j1]`）

- 操作：胜利次数 `ans++`
- 指针移动：`i1++`（我方弱指针后移，下一个弱元素）、`j1++`（对方弱指针后移）
- 逻辑：用最弱元素赢对方最弱，保留我方强元素应对更优场景

### 情况2：我方最弱 < 对方最弱（`a[i1] < b[j1]`）

#### 子情况2.1：我方最强 > 对方最强（`a[i2] > b[j2]`）

- 操作：胜利次数`ans++`
- 指针移动：`i2--`（我方强指针前移）、`j2--`（对方强指针前移）
- 逻辑：用我方最强赢对方最强，确保拿下该轮胜利

#### 子情况2.2：我方最强 < 对方最强

- 操作：无胜利计数
- 指针移动：`i1++`（我方弱指针后移）、`j2--`（对方强指针前移）
- 逻辑：用我方最弱元素消耗对方最强元素，避免我方强元素被对方最强元素浪费

## 三、循环终止条件

当 `i1 > i2` 时，说明我方所有元素已完成匹配，循环结束

## 四、输出结果

打印最终胜利次数 `ans`

## 【高频易错点提醒】

- ❌ 循环条件错误：切勿用跨数组指针比较（如 `i1 <= j2`），会导致提前终止或无限循环
- ❌ 数组冗余：输入时若已初始化数组长度（如 `vector<int> a(n)`），别用 `push_back` 追加元素，避免数组长度翻倍
- ✅ 无平局优化：题目明确无平局时，可直接省略平局处理分支，简化代码

# kmp

```cpp
#include <iostream>
#include <vector>
#include <string>

using namespace std;

// 【战略天赋】Next 数组生成
// 这里的 next[i] 表示：子串 P[0...i] 的最长相等前后缀长度
vector<int> buildNext(const string& P) {
    int m = P.length();
    vector<int> next(m, 0); // 初始化全为0
    
    // j 代表前缀的末尾位置，也代表当前【最长相等前后缀的长度】
    // i 代表后缀的末尾位置
    for (int i = 1, j = 0; i < m; i++) {
        // 核心：如果不匹配，j 就听从 next 数组的指挥，回退寻找更短的匹配
        // 这就是“绝境爆发”前的蓄力，不断回退直到找到支点或者归零
        while (j > 0 && P[i] != P[j]) {
            j = next[j - 1]; // 这里的 j-1 是精髓，回退到上一个匹配状态
        }
        
        // 如果匹配成功，长度 +1
        if (P[i] == P[j]) {
            j++;
        }
        
        // 记录战果
        next[i] = j;
    }
    return next;
}

// 【主战场】KMP 匹配
void kmpSearch(const string& S, const string& P) {
    int n = S.length();
    int m = P.length();
    
    if (m == 0) return;

    // 1. 获取情报
    vector<int> next = buildNext(P);
    
    // 2. 开始扫描
    // i 扫描主串 S (永不回退!)
    // j 扫描模式串 P
    for (int i = 0, j = 0; i < n; i++) {
        // 如果遇到不匹配，模式串指针 j 利用 next 数组战术后撤
        while (j > 0 && S[i] != P[j]) {
            j = next[j - 1];
        }
        
        // 匹配成功，j 前进一步
        if (S[i] == P[j]) {
            j++;
        }
        
        // 判断是否完全通关
        if (j == m) {
            cout << "Found pattern at index: " << (i - m + 1) << endl;
            // 找到一个后，继续寻找下一个，假装最后一位不匹配，回退 j
            j = next[j - 1]; 
        }
    }
}

int main() {
    string S = "ABABABABCABA";
    string P = "ABABC";
    
    cout << "主串: " << S << endl;
    cout << "模式串: " << P << endl;
    
    kmpSearch(S, P);
    
    return 0;
}
```

# 字符串哈希（前缀哈希法）

```cpp
#include <iostream>
using namespace std;

// 1. 类型与常量定义
typedef unsigned long long ULL;  // 无符号长整型：利用自然溢出实现自动取模
const int N = 100010, P = 131;   // N：字符串最大长度；P：哈希基数（常用131/13331）

// 全局变量
int n, m;          // n：字符串长度；m：查询次数
char str[N];       // 存储字符串（下标从1开始，方便前缀计算）
ULL h[N], p[N];    // h[i]：前i个字符的前缀哈希值；p[i]：基数P的i次幂（P^i）

// 2. 子串哈希值计算函数
ULL get(int l, int r) {
    // 公式：子串str[l..r]的哈希值 = h[r] - h[l-1] * P^(r-l+1)
    return h[r] - h[l - 1] * p[r - l + 1];
}

// 3. 主函数（预处理+查询）
int main() {
    // 输入字符串（str+1表示从下标1开始存储）
    scanf("%d%d%s", &n, &m, str + 1);
    
    // 预处理：初始化p数组 + 计算前缀哈希h数组
    p[0] = 1;  // P^0 = 1
    for (int i = 1; i <= n; i++) {
        p[i] = p[i - 1] * P;          // 递推计算P^i
        h[i] = h[i - 1] * P + str[i]; // 递推计算前i个字符的哈希值
    }
    
    // 处理m次查询
    while (m--) {
        int l1, r1, l2, r2;
        scanf("%d%d%d%d", &l1, &r1, &l2, &r2);
        // 比较两个子串的哈希值，相等则子串相同
        if (get(l1, r1) == get(l2, r2)) puts("Yes");
        else puts("No");
    }
    return 0;
}
```

# BFS

```c++
#include <iostream>
#include <cstring>
#include <queue>  // 必须包含的头文件，queue的核心头文件
using namespace std;

typedef pair<int, int> PII;
const int N = 110;  // 网格大小，题目给多大改多大，不影响queue
int g[N][N];        // 网格：0可走，1障碍物
int d[N][N];        // 步数数组，-1表示未访问
int n, m;
// 固定方向数组：上下左右，永远不变
int dx[4] = {-1, 0, 1, 0}, dy[4] = {0, 1, 0, -1};

int bfs()
{
    queue<PII> q;  // 定义STL队列，存坐标(x,y)，固定写法
    
    // 初始化：起点入队 + 步数初始化
    memset(d, -1, sizeof d);
    d[0][0] = 0;
    q.push({0, 0});  // 入队：起点(0,0)，固定写法

    // BFS主循环：队列不空就处理，固定判断条件
    while (!q.empty())
    {
        auto t = q.front();  // 取队头元素，固定写法
        q.pop();             // 队头出队，固定写法！必须写，否则队列永远不空

        // 遍历4个方向
        for (int i = 0; i < 4; i++)
        {
            int x = t.first + dx[i], y = t.second + dy[i];
            // 合法性判断：坐标合法+可走+未访问，缺一不可
            if (x >= 0 && x < n && y >= 0 && y < m && g[x][y] == 0 && d[x][y] == -1)
            {
                d[x][y] = d[t.first][t.second] + 1;
                q.push({x, y});  // 新坐标入队，固定写法
            }
        }
    }
    // 返回终点步数
    return d[n-1][m-1];
}

int main()
{
    cin >> n >> m;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> g[i][j];
    cout << bfs() << endl;
    return 0;
}
```

# dijksta

```cpp
#include <stdio.h>
#include <string.h>
#include <algorithm>
using namespace std;

// ========================== 全局变量定义 + 注释 ==========================
// 算法适用：求【单源最短路径】，即从一个起点到图中所有其他节点的最短距离
// 核心要求：图中所有边的权值 必须是非负数，不能有负权边！
const int N = 1010;  // 定义图的最大节点数，根据题目需求调整，一般1010够用
const int INF = 0x3f3f3f3f;  // 代表无穷大，这个值的好处：1.数值足够大 2.相加不会溢出 3.memset可以直接赋值
int g[N][N];  // 邻接矩阵：g[a][b] 表示 节点a 到 节点b 的直接边的权值
int dist[N];  // 距离数组：dist[j] 表示 起点(这里固定是1号节点) 到 节点j 的当前最短距离
bool st[N];   // 标记数组：st[j] = true 代表 节点j的最短路径已经确定，后续无需再更新
int n, m;     // n = 图的节点总数，m = 图的边的总数

// ========================== Dijkstra核心算法函数 ==========================
// 函数功能：计算从 1号起点 到 n号终点 的最短路径长度，返回该长度
// 核心思想：贪心策略 - 每次选出「当前未确定最短路径、且离起点最近」的节点，用它去优化其他节点的距离
// 因为无负权边，一旦某个节点的最短路径确定，就永远不会被更新，这是贪心的正确性保证
int dijkstra()
{
    // 1. 初始化距离数组：所有节点到起点的距离初始化为无穷大
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;  // 起点(1号节点)到自己的距离，固定为0
    
    // 2. 外层循环n次：因为图中有n个节点，每个节点都要确定一次最短路径
    for (int i = 0; i < n; i++)
    {
        // 2.1 第一步：找到当前 未被确定最短路径(!st[t]) 且 距离起点最近 的节点t
        int t = -1;  // 初始化t为无效节点
        for (int j = 1; j <= n; j++)
        {
            // 筛选条件：j节点未确定 + (t是无效节点 或 j的距离比t更小)
            if (!st[j] && (t == -1 || dist[j] < dist[t]))
                t = j;
        }
        
        // 2.2 第二步：标记该节点t，确认它的最短路径已经敲定，后续不再处理
        st[t] = true;
        
        // 2.3 第三步：核心操作 - 松弛所有节点
        // 用刚确定的节点t，作为中转站，更新起点到其他所有节点j的最短距离
        // 松弛公式：起点→j 的距离  = min(原来的距离, 起点→t的距离 + t→j的距离)
        for (int j = 1; j <= n; j++)
        {
            dist[j] = min(dist[j], dist[t] + g[t][j]);
        }
    }
    
    // 3. 最终判断：如果终点n的距离还是无穷大，说明 起点1 到 终点n 无路径可达
    if (dist[n] == INF) return -1;
    // 否则返回 起点1到终点n的最短路径长度
    return dist[n];
}

// ========================== 主函数 - 数据输入 + 算法调用 ==========================
int main()
{
    // 1. 输入节点数 和 边数
    scanf("%d%d", &n, &m);
    
    // 2. 初始化邻接矩阵：所有边初始化为无穷大，表示初始状态下任意两点之间无直接边
    memset(g, 0x3f, sizeof g);
    
    // 3. 输入m条边的信息：每条边包含 起点a、终点b、边权c
    while (m--)
    {
        int a, b, c;
        scanf("%d%d%d", &a, &b, &c);
        // 处理【重边】问题：如果a→b有多个不同权值的边，只保留权值最小的那条
        // 原因：求最短路径时，大的权值边不可能成为最优解，留最小的即可
        g[a][b] = min(g[a][b], c);
    }
    
    // 4. 调用Dijkstra算法，接收返回的最短路径长度
    int res = dijkstra();
    
    // 5. 输出结果
    printf("%d\n", res);
    
    return 0;
}

// ========================== 补充重要笔记【必看】==========================
/*
1. 时间复杂度：O(n²)，n是节点数，适合 n<=1000 的数据范围，效率足够
2. 邻接矩阵优缺点：优点-代码简单、容易理解、适合初学者入门；缺点-空间复杂度O(n²)，节点多了会超内存
3. 关于无穷大0x3f3f3f3f：十进制是1061109567，这个数比题目中常见的最大路径和大，且两个相加不会超int范围
4. 节点编号：本代码节点编号从1开始，是算法题的通用写法，比从0开始更符合题意
5. 拓展：如果题目节点数 n>1000（比如n=1e5），需要用【邻接表+堆优化】的Dijkstra，时间复杂度O(m logn)
6. 易错点：忘记处理重边、有负权边还强行用Dijkstra、初始化距离数组时没把起点设为0
*/
```

# bellman-ford

# 带边数限制的Bellman-Ford 最短路径算法 - 完整注释版代码笔记
## ✅ 算法核心说明
**适用场景**：求解**从指定起点出发，经过最多k条边**的单源最短路径问题（本题固定起点为节点1，终点为节点n）
**核心优势**：可以处理图中存在**负权边**的情况，这是迪杰斯特拉(Dijkstra)算法做不到的
**核心特点**：本版实现为Bellman-Ford的经典应用，只解决「边数限制+最短路径」问题，未做负权回路检测（边数有限时，负权回路无影响）
**核心思想**：对所有边进行最多`k`轮松弛操作，每一轮松弛对应**多走1条边**，严格限制路径的边数不超过k

---

## ✅ 完整带详细注释的C语言代码（可直接运行/复制）
~~~cpp

#include <stdio.h>
#include <string.h>
#include <algorithm>
using namespace std;

// ====================== 全局宏定义 - 刷题通用配置 ======================
// 0x3f3f3f3f 是C语言中常用的无穷大值，满足两个条件：
// 1. 数值足够大，大于图论中所有可能的路径权值和
// 2. 是int范围内的数，不会溢出，且 INF/2 也能正常使用
const int INF = 0x3f3f3f3f;
// 最大边数和最大节点数，根据题目要求调整即可，刷题常用取值
const int M = 100010;  // 边的最大数量
const int N = 510;     // 节点的最大数量

// ====================== 数据结构定义 ======================
// 边的结构体：Bellman-Ford算法采用【边集数组】存储所有边即可，无需邻接表/邻接矩阵
struct Edge
{
    int a;  // 本条边的起点节点编号
    int b;  // 本条边的终点节点编号
    int w;  // 本条边的权值（可以是负数，支持负权边）
}edges[M];   // 定义边集数组，存储整张图的所有边

// ====================== 全局数组定义 ======================
int dist[N];   // dist[x] 表示：从起点(节点1)到节点x的「当前最短距离」
int backup[N]; // 备份数组：备份上一轮松弛后的dist数组，核心作用看函数内注释
int n, m, k;   // 全局变量：n=节点总数，m=边总数，k=最多经过的边数

// ====================== 核心算法：带边数限制的Bellman-Ford ======================
// 函数功能：计算从节点1 到 节点n 最多经过k条边的最短路径
// 返回值：成功返回最短路径权值，不可达返回-1
int bellman_ford()
{
    // 1. 初始化距离数组：所有节点到起点的距离初始化为无穷大
    memset(dist, INF, sizeof dist);
    dist[1] = 0; // 起点是节点1，节点1到自身的距离为0，这是松弛的初始条件

    // 2. 核心循环：循环k次 = 最多经过k条边
    // 每一轮循环，都代表「从起点出发，多走1条边」的松弛操作
    for(int i = 0; i < k; i++)
    {
        // 【重中之重】每次松弛前，必须备份上一轮的dist数组到backup
        // 作用：保证本轮所有边的松弛操作，都基于「上一轮的最短距离」进行更新
        // 防止出现：同一次循环中，刚松弛更新的节点，立刻参与下一条边的松弛，导致「本轮走了多条边」
        // 简单说：backup的存在，严格限制了 每轮循环只走1条边，最终最多走k条边
        memcpy(backup, dist, sizeof dist);

        // 3. 遍历整张图的所有边，对每条边执行【松弛操作】
        for(int j = 0; j < m; j++)
        {
            // 取出当前遍历的这条边：起点a、终点b、权值w
            int a = edges[j].a;
            int b = edges[j].b;
            int w = edges[j].w;

            // 松弛公式：尝试用 起点a的最短距离 + 边权w，更新 终点b的最短距离
            // 这里必须用 backup[a] 而不是 dist[a]，就是上面说的备份的意义
            dist[b] = min(dist[b], backup[a] + w);
        }
    }

    // 4. 判断终点是否可达：为什么判断 > INF/2 而不是 == INF？
    // 原因：图中存在负权边时，某些不可达节点的距离会被负权边「轻微更新」，变成 小于INF但依然很大的数
    // 0x3f3f3f3f/2 是安全阈值，大于这个值就认为是「不可达」
    if(dist[n] > INF / 2)
        return -1;
    else
        return dist[n]; // 可达则返回最短路径权值
}

// ====================== 主函数：输入数据 + 调用算法 + 输出结果 ======================
int main()
{
    // 输入：节点数n、边数m、最多经过的边数k
    scanf("%d%d%d", &n, &m, &k);
    // 循环输入m条边的信息：起点a、终点b、权值w
    for(int i = 0; i < m; i++)
    {
        int a, b, w;
        scanf("%d%d%d", &a, &b, &w);
        edges[i] = {a, b, w}; // 存入边集数组
    }

    // 调用核心算法，接收返回结果
    int ans = bellman_ford();
    // 输出结果：返回-1则输出impossible，否则输出最短距离
    if(ans == -1)
        printf("impossible\n");
    else
        printf("%d\n", ans);

    return 0;
}

## ✅ 核心知识点笔记（必记，重中之重）
### 1. backup数组 为什么是核心？能不能省略？
**绝对不能省略**，这是「带边数限制」的Bellman-Ford算法最核心的细节，也是高频面试/笔试考点
- 省略backup：算法会退化为普通Bellman-Ford，能求最短路径，但**无法限制边数**，会出现「一轮循环走多条边」的情况，违反题目要求
- 保留backup：严格保证 **每一轮松弛，都只能基于上一轮的距离结果更新**，每轮循环等价于「只能多走1条边」，循环k次就等价于「最多走k条边」

### 2. 无穷大值 `0x3f3f3f3f` 的优势
本代码中用的`0x3f3f3f3f` 是C语言图论题的**最优无穷大取值**，对比其他取值的优势：
1. 数值大小：`0x3f3f3f3f = 1061109567`，是int范围内的数，不会溢出
2. 赋值方便：可以用`memset(dist, INF, sizeof dist)` 批量赋值（memset按字节赋值，0x3f的每个字节都是相同的）
3. 容错性好：判断不可达时用 `dist[n] > INF/2`，可以规避负权边对「无穷大」的轻微修改

### 3. 松弛操作的本质
`dist[b] = min(dist[b], backup[a] + w)`
> 对于边 `a -> b`，如果「从起点走到a的最短距离 + a到b的边权」 比 「当前从起点走到b的最短距离」更小，就更新b的最短距离。
> 这是所有最短路径算法的核心思想（Dijkstra、SPFA、Floyd 都基于松弛操作）

### 4. 算法时间复杂度
`O(k * m)` ：循环k次，每次循环遍历m条边，这是固定复杂度
- 优势：k很小的时候，算法效率极高
- 劣势：k和m都很大时，效率偏低，但这是该场景下的最优解
~~~

```cpp
#include <stdio.h>
#include <cstring>
#include <queue>
using namespace std;

// ========== 纯数组模拟邻接表 固定模板 (你学的版本，一字不改) ==========
const int N = 1005;   // 节点最大数量，题目给多少改多少
const int M = 2005;   // 边最大数量，一般是节点数的2倍
const int INF = 0x3f3f3f3f; // 无穷大，固定值

int h[N];  // h[u]：节点u的第一条边的下标
int e[M];  // e[idx]：编号idx的边 指向的邻点
int ne[M]; // ne[idx]：编号idx的边 的下一条边下标
int w[M];  // w[idx]：编号idx的边 的权值
int idx;   // 边的计数器，初始为0

// 加边函数：添加 有向边 u→v，权值val (固定模板)
void add(int u, int v, int val) {
    e[idx] = v;
    w[idx] = val;
    ne[idx] = h[u];
    h[u] = idx++;
}

// ========== SPFA核心数组 ==========
int dist[N];  // dist[u]：起点到u的最短距离
bool st[N];   // st[u]：节点u是否在队列中 (弹出必置false，核心规则)
int cnt[N];   // ✅ 核心：存【起点到u的最短路径的边数】，你的最优写法！

// ========== SPFA核心函数 ==========
// start：起点编号，n：总节点数
// 返回值：true=存在负权回路  false=无负权回路
bool spfa(int start, int n) {
    // 初始化所有数组
    memset(dist, INF, sizeof dist);
    memset(st, false, sizeof st);
    memset(cnt, 0, sizeof cnt); // 路径边数初始化为0
    queue<int> q;

    // 起点初始化
    dist[start] = 0;
    q.push(start);
    st[start] = true;
    cnt[start] = 0; // 起点到自己，路径边数为0，必须写！

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        st[u] = false; // ✅ 弹出队头，立刻取消标记，SPFA铁律！

        // 遍历u的所有邻边 (数组邻接表固定遍历方式，一字不改)
        for (int i = h[u]; i != -1; i = ne[i]) {
            int v = e[i];   // u的邻点
            int val = w[i]; // u→v的边权

            // 松弛操作：核心公式，所有最短路径算法都不变
            if (dist[v] > dist[u] + val) {
                dist[v] = dist[u] + val;
                cnt[v] = cnt[u] + 1;  // ✅ 你的封神优化！存路径边数！
                
                // ✅ 立刻检测负权回路，无多余循环，效率最高！
                if (cnt[v] >= n) {
                    return true;
                }

                // 节点不在队列中，才入队，防重复入队
                if (!st[v]) {
                    q.push(v);
                    st[v] = true;
                }
            }
        }
    }
    // 无负权回路，正常结束
    return false;
}

// ========== 测试你的图 (1→2→4↔3 负权回路) ==========
int main() {
    // 邻接表初始化：头指针数组置-1，边计数器置0
    memset(h, -1, sizeof h);
    idx = 0;

    int n = 4; // 你的图共4个节点
    // 加边：1→2(2)  2→4(3)  4→3(-1)  3→4(-1)
    add(1, 2, 2);
    add(2, 4, 3);
    add(4, 3, -1);
    add(3, 4, -1);

    // 运行SPFA，起点为1
    bool has_neg_cycle = spfa(1, n);
    if (has_neg_cycle) {
        printf("✅ 检测到负权回路！\n");
    } else {
        printf("图中无负权回路\n");
    }
    // 输出正确的最短距离
    printf("起点1到终点4的最短距离 = %d\n", dist[4]);
    return 0;
}
```

~~~md
# 树链剖分 + 树上差分 解决“删边使所有点对不连通”问题
## 一、问题背景
给定一棵有 `n` 个节点的树（无环、任意两点只有唯一一条路径），以及 `m` 个无序点对 `(a₁,b₁), (a₂,b₂)...(aₘ,bₘ)`。要求找到一条边，满足：
- 删除这条边后，所有 `m` 个点对都不再连通（每个点对的两个节点被分到两个子树）；
- 若有多个符合条件的边，选择编号最大的；
- 若无符合条件的边，输出 `-1`。

## 二、核心知识点1：树链剖分（重链剖分）
### 2.1 树链剖分核心概念（新手友好版）
树链剖分的核心目标是**将复杂的树拆解为若干条“链”**，从而高效处理“两点路径”相关问题（如找最近公共祖先LCA、统计路径权重等）。关键概念如下：

| 概念         | 通俗解释                                                                 |
|--------------|--------------------------------------------------------------------------|
| 重儿子       | 一个节点的所有子节点中，**子树节点数量最多**的子节点（“最核心”的子节点）。 |
| 轻儿子       | 除重儿子外的其他子节点。                                                 |
| 重链         | 由“父节点 + 重儿子”依次连接形成的连续链（如：根→重儿子→重孙子…）。       |
| 轻链         | 轻儿子自身作为链顶形成的短链（通常仅包含1个节点）。                       |
| 链顶（top）  | 每条链的起始节点（重链链顶为链的第一个节点，轻链链顶为轻儿子自身）。       |
| 子树大小（siz） | 以某个节点为根的子树包含的总节点数。                                     |
| 深度（dep）   | 节点到根节点的距离（根节点深度为1）。                                     |
| 父节点（fa）  | 节点的直接上级节点（根节点父节点为0）。                                   |

### 2.2 树链剖分的两次DFS预处理
树链剖分需通过两次DFS完成核心信息的预处理，为后续操作打基础。

#### （1）第一次DFS（dfs1）：贴基础标签
```cpp
void dfs1(int u,int father)  // u：当前节点，father：u的父节点
{
  siz[u]=1;                  // 初始：当前节点自身计为1个节点
  dep[u]=dep[father]+1;      // 深度 = 父节点深度 + 1（根节点father=0，dep[1]=1）
  fa[u]=father;              // 记录当前节点的父节点
  
  // 遍历当前节点的所有邻居
  for(int i=0;i<edge[u].size();i++)
  {
    int s=edge[u][i];        // s是u的一个邻居节点
    if(s==fa[u]) continue;   // 跳过父节点，避免回头遍历导致死循环
    
    dfs1(s,u);               // 递归处理子节点s（父节点为u）
    siz[u]+=siz[s];          // 汇总子节点的子树大小到当前节点
    
    // 更新重儿子：选择子树最大的子节点作为重儿子
    if(siz[son[u]]<siz[s]) son[u]=s;
  }
}
```
**核心作用**：计算每个节点的 `siz`（子树大小）、`dep`（深度）、`fa`（父节点）、`son`（重儿子）。
**递归逻辑**：后序遍历（先处理所有子节点，再汇总到父节点），确保子树大小统计完整。

#### （2）第二次DFS（dfs2）：标记链顶（完成剖分）
```cpp
void dfs2(int u,int t)  // u：当前节点，t：u所在链的链顶
{
  top[u]=t;              // 标记当前节点的链顶
  if(son[u]==0) return;  // 无重儿子（叶子节点），直接返回
  
  dfs2(son[u],t);        // 处理重儿子：链顶不变，保持重链连续
  
  // 处理所有轻儿子
  for(int i=0;i<edge[u].size();i++)
  {
    int s=edge[u][i];
    // 跳过父节点、跳过重儿子（已处理）
    if(s==fa[u]||s==son[u]) continue;
    
    dfs2(s,s);           // 轻儿子自身作为新链的链顶（形成轻链）
  }
}
```
**核心作用**：给每个节点标记 `top`（链顶），完成树的“链剖分”。
**递归逻辑**：先重后轻（优先处理重儿子保证重链连续，再处理轻儿子形成轻链）。

### 2.3 树链剖分找LCA（最近公共祖先）
LCA是两个节点向上遍历的第一个公共节点，树链剖分通过“跳链”快速查找LCA：
```cpp
int lca(int x,int y)
{
  // 若x和y不在同一条链上，持续“跳链”
  while(top[x]!=top[y])
  {
    // 让链顶深度更大的节点向上跳（保证方向正确）
    if(dep[top[x]]<dep[top[y]]) swap(x,y);
    x=fa[top[x]];  // 跳到当前链顶的父节点（等价于“跳过整条链”）
  }
  // 同链后，深度更小的节点即为LCA
  return dep[x]<dep[y] ? x : y;
}
```
**查找步骤（通俗版）**：
1. 若x、y不在同链：将“链顶深度更大”的节点跳到其链顶的父节点；
2. 重复步骤1，直到x、y同链；
3. 同链后，深度更小的节点就是LCA。

**示例**：树结构为 `1-2-3`（1为根）
- 查找3和1的LCA：top[3]=1、top[1]=1（同链），dep[1]<dep[3] → LCA=1；
- 查找3和2的LCA：top[3]=1、top[2]=1（同链），dep[2]<dep[3] → LCA=2。

## 三、核心知识点2：树上差分
### 3.1 树上差分的目的
避免逐个遍历点对路径（大数据下超时），通过“标记路径端点+LCA”，再汇总的方式，**高效统计每条边被点对路径经过的次数**。

### 3.2 差分标记规则
对每个点对 `(a,b)`，找到其LCA后执行以下操作：
```cpp
w[a]++;     // 起点a的权重+1
w[b]++;     // 终点b的权重+1
w[lca(a,b)]-=2;  // LCA的权重-2
```
**原理**：标记 `a→LCA` 和 `b→LCA` 的路径，汇总后每个节点的`w`值对应“该节点到父节点的边”的经过次数。

### 3.3 差分汇总（lcasum函数）
通过后序遍历汇总子节点权重，得到每条边的最终经过次数：
```cpp
void lcasum(int u,int father)
{
  for(int i=0;i<edge[u].size();i++)
  {
    int v=edge[u][i];        // v是u的邻居节点
    if(v==father) continue;  // 跳过父节点
    
    lcasum(v,u);             // 先递归处理子节点
    w[u]+=w[v];              // 子节点权重汇总到父节点
  }
}
```
**关键结论**：汇总后，`w[i]` 的值 = “节点i到其父节点fa[i]的边”被点对路径经过的次数。

## 四、完整代码（带详细注释）
```cpp
#include <bits/stdc++.h>
using namespace std;
typedef pair<int,int>pii;
const int N=1e5+9;  // 数组最大长度，适配1e5级别的节点规模

// 全局变量定义
int n,m;            // n=节点数，m=点对数
int w[N];           // 差分权重数组（全局变量默认初始化为0）
map<pii,int>id;     // 存储边编号：键为(节点x,节点y)，值为边编号（双向存储）
vector<int>edge[N]; // 邻接表：edge[u]存储u的所有邻居节点

// 树链剖分核心数组
int siz[N];  // siz[u]：以u为根的子树节点数
int dep[N];  // dep[u]：u的深度
int fa[N];   // fa[u]：u的父节点
int son[N];  // son[u]：u的重儿子
int top[N];  // top[u]：u所在链的链顶

// 第一次DFS：预处理siz/dep/fa/son
void dfs1(int u,int father)
{
  siz[u]=1;                  // 初始子树大小为1（仅包含自身）
  dep[u]=dep[father]+1;      // 深度继承父节点并+1
  fa[u]=father;              // 记录父节点
  
  for(int i=0;i<edge[u].size();i++)
  {
    int s=edge[u][i];        // 遍历当前节点的邻居
    if(s==fa[u]) continue;   // 跳过父节点，避免回环
    
    dfs1(s,u);               // 递归处理子节点
    siz[u]+=siz[s];          // 汇总子节点的子树大小
    
    // 更新重儿子：选择子树最大的子节点
    if(siz[son[u]]<siz[s]) son[u]=s;
  }
}

// 第二次DFS：标记链顶，完成树链剖分
void dfs2(int u,int t)
{
  top[u]=t;                  // 标记当前节点的链顶
  if(son[u]==0) return;      // 无重儿子（叶子节点），直接返回
  
  dfs2(son[u],t);            // 处理重儿子：链顶保持不变
  
  // 处理所有轻儿子
  for(int i=0;i<edge[u].size();i++)
  {
    int s=edge[u][i];
    if(s==fa[u]||s==son[u]) continue;
    
    dfs2(s,s);               // 轻儿子自身作为新链的链顶
  }
}

// 树链剖分查找LCA
int lca(int x,int y)
{
  // 不同链则持续跳链
  while(top[x]!=top[y])
  {
    // 链顶深度更大的节点向上跳
    if(dep[top[x]]<dep[top[y]]) swap(x,y);
    x=fa[top[x]];
  }
  // 同链后，深度更小的节点为LCA
  return dep[x]<dep[y] ? x : y;
}

// 差分汇总：后序遍历计算每条边的经过次数
void lcasum(int u,int father)
{
  for(int i=0;i<edge[u].size();i++)
  {
    int v=edge[u][i];
    if(v==father) continue;
    
    lcasum(v,u);             // 先处理子节点
    w[u]+=w[v];              // 汇总子节点的权重
  }
}

// 核心业务逻辑
void solve()
{
  cin>>n>>m;
  
  // 建边：存储邻接表 + 边编号
  for(int i=1;i<=n-1;i++)
  {
    int x,y;cin>>x>>y;
    edge[x].push_back(y);
    edge[y].push_back(x);
    id[{x,y}]=id[{y,x}]=i;  // 双向存储边编号，方便后续查找
  }
  
  // 树链剖分预处理
  dfs1(1,0);  // 根节点为1，父节点为0
  dfs2(1,1);  // 根节点的链顶为自身
  
  // 处理m个点对：执行差分标记
  for(int i=0;i<m;i++)
  {
    int a,b;cin>>a>>b;
    w[a]++;
    w[b]++;
    w[lca(a,b)]-=2;
  }
  
  // 汇总差分结果，得到每条边的经过次数
  lcasum(1,0);
  
  // 查找答案：被所有m个点对经过的最大编号边
  int ans=-1;
  for(int i=1;i<=n;i++)
  {
    // w[i]==m 说明i到fa[i]的边被所有点对路径经过
    if(w[i]==m)
    {
      ans=max(ans,id[{i,fa[i]}]);
    }
  }
  cout<<ans<<endl;
}

// 程序入口
int main()
{
  solve();
  return 0;
}
```

## 五、核心逻辑总结
### 5.1 整体执行流程
1. **建边**：用邻接表存储树结构，用map存储边编号（双向）；
2. **树链剖分**：两次DFS预处理，得到找LCA所需的所有信息；
3. **差分标记**：对每个点对(a,b)，标记a、b、LCA的权重；
4. **差分汇总**：后序遍历汇总权重，得到每条边的经过次数；
5. **查找答案**：遍历所有节点，选择“经过次数=m”的最大编号边。

### 5.2 关键结论
1. 树链剖分的核心是“拆树为链”，核心目标是高效查找LCA；
2. 树上差分的核心是“标记路径端点+LCA，汇总得边的经过次数”；
3. `w[i]==m` 是判断“i→fa[i]的边”符合条件的核心条件；
4. 最终答案选择符合条件的最大边编号，无则输出-1。

### 5.3 新手注意事项
1. 根节点1的父节点为0，遍历节点时无需特殊处理（测试用例中w[1]不会等于m）；
2. map存储边编号适合小规模数据，大数据场景建议替换为数组`edge_id[N]`（存储节点到父节点的边编号）；
3. 树链剖分的两次DFS顺序不可颠倒（需先执行dfs1贴基础标签，再执行dfs2剖分）。
~~~
